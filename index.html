import http.server import socketserver import webbrowser import threading import time import json import serial import serial.tools.list_ports from threading import Thread class CarController: def __init__(self): self.port = None self.is_connected = False self.running = False def connect(self): try: ports = serial.tools.list_ports.comports() print("\n可用串口:") for i, p in enumerate(ports): print(f" [{i}] {p.device} - {p.description}") selected = 'COM4' for p in ports: if '传出' in p.description or 'Outgoing' in p.description: selected = p.device break self.port = serial.Serial(port=selected, baudrate=9600, timeout=1, write_timeout=2) self.is_connected = True self.running = True Thread(target=self._read, daemon=True).start() print(f"✅ 连接成功: {selected}") return True except Exception as e: print(f"❌ 连接失败: {e}") return False def _read(self): while self.running and self.port and self.port.is_open: try: if self.port.in_waiting > 0: data = self.port.readline().decode('utf-8', errors='ignore').strip() if data: print(f"← {data}") time.sleep(0.05) except: break def send(self, cmd): if not self.is_connected or not self.port: return False try: self.port.write(cmd.encode('utf-8')) print(f"→ {cmd}") return True except: return False def disconnect(self): self.running = False if self.port and self.port.is_open: self.port.close() self.is_connected = False print("🔌 已断开") car = CarController() HTML = ''' 智能小车控制器

智能小车控制器

蓝牙遥控 · 自动巡线
未连接
模式
方向控制（按住移动，松开停止）
速度
''' class Handler(http.server.SimpleHTTPRequestHandler): def do_GET(self): if self.path == '/': self.send_response(200) self.send_header('Content-type', 'text/html; charset=utf-8') self.end_headers() self.wfile.write(HTML.encode('utf-8')) else: self.send_response(404) self.end_headers() def do_POST(self): if self.path == '/api/connect': self._json({'ok': car.connect()}) elif self.path == '/api/disconnect': car.disconnect() self._json({'ok': True}) elif self.path == '/api/command': data = json.loads(self.rfile.read(int(self.headers['Content-Length']))) self._json({'ok': car.send(data.get('command', ''))}) else: self.send_response(404) self.end_headers() def _json(self, obj): self.send_response(200) self.send_header('Content-type', 'application/json') self.end_headers() self.wfile.write(json.dumps(obj).encode('utf-8')) def log_message(self, f, *a): print(f"[HTTP] {self.command} {self.path}") if __name__ == "__main__": PORT = 8000 threading.Thread(target=lambda: (time.sleep(1), webbrowser.open(f'http://localhost:{PORT}')), daemon=True).start() print(f"🚀 服务器已启动: http://localhost:{PORT}") print("📱 在浏览器中操控小车") with socketserver.TCPServer(("", PORT), Handler) as h: try: h.serve_forever() except KeyboardInterrupt: print("\n已停止")
