import http.server import socketserver import webbrowser import threading import time import json from datetime import datetime # ==================== 前端界面（含真实蓝牙逻辑） ==================== HTML_CONTENT = ''' 智能小车遥控系统

智能小车遥控系统

请点击下方按钮连接小车蓝牙

未连接


小车状态: 离线

电池电量: --

''' # ==================== 后端服务器 ==================== class CarControlHTTPRequestHandler(http.server.SimpleHTTPRequestHandler): def do_GET(self): if self.path == '/': self.send_response(200) self.send_header('Content-type', 'text/html; charset=utf-8') self.end_headers() self.wfile.write(HTML_CONTENT.encode('utf-8')) elif self.path == '/api/status': self.send_response(200) self.send_header('Content-type', 'application/json') self.end_headers() # 修正点4：返回真实连接状态（与前端同步） status = { "connected": False, # 初始未连接 "speed": 50, "direction": "stop", "battery": 85, "timestamp": datetime.now().isoformat() } self.wfile.write(json.dumps(status).encode('utf-8')) else: self.send_response(404) self.end_headers() self.wfile.write(b'404 Not Found') def start_server(port=8000): with socketserver.TCPServer(("", port), CarControlHTTPRequestHandler) as httpd: print(f"🚀 服务器启动: http://localhost:{port}") print("⚠️ 需用Chrome/Edge打开（支持Web Bluetooth）") print("⚠️ 请替换代码中的蓝牙UUID为你的小车真实服务ID") try: httpd.serve_forever() except KeyboardInterrupt: print("\n🛑 服务器已停止") if __name__ == "__main__": # 自动打开浏览器 threading.Thread( target=lambda: (time.sleep(1), webbrowser.open('zhu200701.github.io ')), daemon=True ).start() start_server()
